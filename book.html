Python ऑटोमेशन स्क्रिप्ट (PyMuPDF / fitz)

यदि आपके पास 100+ किताबों की बड़ी बल्क फाइलें हैं तो इस पायथन स्क्रिप्ट का प्रयोग करें:

import fitz  # PyMuPDF पैकेज संस्थापित करें: pip install pymupdf

def process_iois_workbook(cover_img_path, orig_pdf_path, output_pdf_path):
    print("IOIS PDF प्रोसेसिंग शुरू हो रही है...")
    
    # 1. नया दस्तावेज़ बनाएं
    final_doc = fitz.open()

    # 2. नया कवर पेज इंसर्ट करें (Page 1)
    img_doc = fitz.open(cover_img_path)
    pdf_bytes = img_doc.convert_to_pdf()
    cover_page_pdf = fitz.open("pdf", pdf_bytes)
    final_doc.insert_pdf(cover_page_pdf)

    # 3. मूल PDF से पेज 2 से अंतिम पेज तक जोड़ें (पहला पुराना कवर हटाकर)
    orig_doc = fitz.open(orig_pdf_path)
    if len(orig_doc) > 1:
        final_doc.insert_pdf(orig_doc, from_page=1, to_page=len(orig_doc) - 1)
    else:
        print("चेतावनी: मूल PDF में केवल 1 पेज था!")

    # 4. हर कंटेंट पेज पर Header और Footer जोड़ें
    header_title = "IOIS INDIAN PUBLICATIONS | बाल विकास - डिजिटल वर्कबुक"
    footer_plans = "IOIS PLATFORM PLANS: BASIC (वर्कबुक) | GOLD (डिजिटल किट) | DIAMOND (स्कूल लाइसेंस)"
    footer_contact = "Website: www.ioisplatform.com | Helpline: +91 98765 43210"

    for page_num in range(1, len(final_doc)): # Cover Page (0) को छोड़कर
        page = final_doc[page_num]
        rect = page.rect

        # (A) Header Background Banner (Navy Blue)
        page.draw_rect(fitz.Rect(0, 0, rect.width, 36), color=None, fill=(0.08, 0.18, 0.36))
        # Header Text
        page.insert_text(fitz.Point(20, 22), header_title, fontsize=10, color=(1, 0.84, 0), fontname="helv-bold")

        # (B) Footer Background Banner (Light Gray)
        page.draw_rect(fitz.Rect(0, rect.height - 36, rect.width, rect.height), color=None, fill=(0.94, 0.95, 0.98))
        # Top Divider Line
        page.draw_line(fitz.Point(0, rect.height - 36), fitz.Point(rect.width, rect.height - 36), color=(0.8, 0.85, 0.9), width=1)
        # Footer Text
        page.insert_text(fitz.Point(15, rect.height - 22), footer_plans, fontsize=7 fontname="helv-bold", color=(0.1, 0.2, 0.4))
        page.insert_text(fitz.Point(15, rect.height - 10), footer_contact, fontsize=7, fontname="helv", color=(0.3, 0.3, 0.3))

    # 5. अंतिम PDF सेव करें
    final_doc.save(output_pdf_path)
    print(f"सफलतापूर्वक जनरेट हो गया: {output_pdf_path}")
